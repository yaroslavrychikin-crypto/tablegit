## Часть 1. ОКОННЫЕ ФУНКЦИИ — расширенные сценарии

### 1.1. Ранжирование: топ-N товаров в каждой категории

```sql
-- Задача: получить 2 самых дорогих товара в каждой категории.
-- Классическая задача, где без оконных функций пришлось бы делать self-join.
WITH Ranked AS (
    SELECT 
        p.ProductID,
        p.Name,
        c.Name AS CategoryName,
        p.Price,
        ROW_NUMBER() OVER (PARTITION BY p.CategoryID ORDER BY p.Price DESC) AS rn
    FROM ref.Products p
    JOIN ref.Categories c ON c.CategoryID = p.CategoryID
)
SELECT * FROM Ranked WHERE rn <= 2 ORDER BY CategoryName, Price DESC;
```

### 1.2. Проценты, доли и накопительные итоги

```sql
-- Задача: по каждому заказу показать сумму позиции, долю от заказа,
-- накопительный итог по заказу и долю от общего оборота.
SELECT 
    oi.OrderID,
    oi.ProductID,
    oi.Quantity * oi.UnitPrice * (1 - oi.Discount/100.0) AS LineTotal,

    -- Доля позиции внутри своего заказа
    oi.Quantity * oi.UnitPrice * (1 - oi.Discount/100.0)
      / SUM(oi.Quantity * oi.UnitPrice * (1 - oi.Discount/100.0)) 
        OVER (PARTITION BY oi.OrderID) * 100 AS PctInOrder,

    -- Накопительный итог по заказу (в порядке OrderItemID)
    SUM(oi.Quantity * oi.UnitPrice * (1 - oi.Discount/100.0))
        OVER (PARTITION BY oi.OrderID ORDER BY oi.OrderItemID
              ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS RunningInOrder,

    -- Доля позиции от общего оборота всех заказов
    oi.Quantity * oi.UnitPrice * (1 - oi.Discount/100.0)
      / SUM(oi.Quantity * oi.UnitPrice * (1 - oi.Discount/100.0)) OVER () * 100 AS PctOfAllSales
FROM sales.OrderItems oi
ORDER BY oi.OrderID, oi.OrderItemID;
```

### 1.3. LAG / LEAD: динамика продаж по месяцам

```sql
-- Задача: помесячный оборот + сравнение с предыдущим и следующим месяцем.
WITH MonthlySales AS (
    SELECT 
        DATEFROMPARTS(YEAR(OrderDate), MONTH(OrderDate), 1) AS MonthStart,
        SUM(TotalAmount) AS Revenue
    FROM sales.Orders
    WHERE Status <> N'отменён'
    GROUP BY DATEFROMPARTS(YEAR(OrderDate), MONTH(OrderDate), 1)
)
SELECT 
    MonthStart,
    Revenue,
    LAG(Revenue)  OVER (ORDER BY MonthStart) AS PrevMonthRevenue,
    LEAD(Revenue) OVER (ORDER BY MonthStart) AS NextMonthRevenue,
    Revenue - LAG(Revenue) OVER (ORDER BY MonthStart) AS DiffVsPrev,
    -- Темп роста в процентах
    CASE WHEN LAG(Revenue) OVER (ORDER BY MonthStart) IS NULL THEN NULL
         ELSE CAST(
            (Revenue - LAG(Revenue) OVER (ORDER BY MonthStart))
            / LAG(Revenue) OVER (ORDER BY MonthStart) * 100 AS DECIMAL(10,2)
         )
    END AS GrowthPct
FROM MonthlySales
ORDER BY MonthStart;
```

### 1.4. FIRST_VALUE / LAST_VALUE с явными рамками

```sql
-- Задача: для каждого сотрудника показать его зарплату, минимальную и максимальную
-- зарплату в его отделе, а также имя самого высокооплачиваемого в отделе.
SELECT 
    EmployeeID,
    FullName,
    Department,
    Salary,
    -- FIRST_VALUE по зарплате DESC = максимум в отделе
    FIRST_VALUE(Salary) OVER (
        PARTITION BY Department 
        ORDER BY Salary DESC
        ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
    ) AS MaxSalaryInDept,
    -- LAST_VALUE по зарплате DESC = минимум в отделе
    LAST_VALUE(Salary) OVER (
        PARTITION BY Department 
        ORDER BY Salary DESC
        ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
    ) AS MinSalaryInDept,
    FIRST_VALUE(FullName) OVER (
        PARTITION BY Department 
        ORDER BY Salary DESC
        ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
    ) AS TopEarner
FROM hr.Employees;
```

### 1.5. NTILE — разбиение клиентов на квартили по обороту

```sql
-- Задача: разделить клиентов на 4 группы по общей сумме покупок.
WITH CustomerRevenue AS (
    SELECT 
        c.CustomerID,
        c.FullName,
        SUM(o.TotalAmount) AS Revenue
    FROM ref.Customers c
    JOIN sales.Orders o ON o.CustomerID = c.CustomerID
    WHERE o.Status <> N'отменён'
    GROUP BY c.CustomerID, c.FullName
)
SELECT 
    FullName,
    Revenue,
    NTILE(4) OVER (ORDER BY Revenue DESC) AS Quartile,
    CASE NTILE(4) OVER (ORDER BY Revenue DESC)
         WHEN 1 THEN N'Платина'
         WHEN 2 THEN N'Золото'
         WHEN 3 THEN N'Серебро'
         ELSE N'Бронза'
    END AS Tier
FROM CustomerRevenue;
```

### 1.6. Скользящее среднее (Moving Average)

```sql
-- Задача: по каждому заказу — сумма и скользящее среднее за последние 3 заказа.
SELECT 
    OrderID,
    OrderDate,
    TotalAmount,
    AVG(TotalAmount) OVER (
        ORDER BY OrderDate, OrderID
        ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
    ) AS MovingAvg3
FROM sales.Orders
WHERE Status <> N'отменён'
ORDER BY OrderDate;
```

### 1.7. Оконные функции в WHERE невозможны — используем CTE

```sql
-- ❌ Так нельзя:
-- SELECT * FROM sales.Orders WHERE ROW_NUMBER() OVER (ORDER BY TotalAmount DESC) = 1;

-- ✅ Правильно:
WITH Numbered AS (
    SELECT *, ROW_NUMBER() OVER (ORDER BY TotalAmount DESC) AS rn
```

### 1.8. WINDOW-клауза (SQL Server 2022+) — многократное использование одного окна

```sql
-- SQL Server 2022 позволяет объявить окно один раз
SELECT 
    OrderID,
    OrderDate,
    TotalAmount,
    SUM(TotalAmount) OVER w AS RunningTotal,
    AVG(TotalAmount) OVER w AS MovingAvg,
    COUNT(*) OVER w AS RunningCount
FROM sales.Orders
WINDOW w AS (
    ORDER BY OrderDate, OrderID
    ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
);
```

### 1.9. PERCENT_RANK, CUME_DIST

```sql
-- Задача: относительное положение каждого сотрудника по зарплате.
SELECT

    Salary,
```

### 2.2. Агрегирующее представление с расчётом маржи

```sql
CREATE OR ALTER VIEW sales.vProductMargin
AS
SELECT 
    p.ProductID,
    p.SKU,
    p.Name,
    cat.Name AS CategoryName,
    p.Price,
    p.CostPrice,
    p.Price - p.CostPrice AS Margin,
    CAST((p.Price - p.CostPrice) / NULLIF(p.Price, 0) * 100 AS DECIMAL(5,2)) AS MarginPct,
    ISNULL(SUM(oi.Quantity), 0) AS TotalSold,
    ISNULL(SUM(oi.Quantity * oi.UnitPrice * (1 - oi.Discount/100.0)), 0) AS Revenue
FROM ref.Products p
JOIN ref.Categories cat ON cat.CategoryID = p.CategoryID
LEFT JOIN sales.OrderItems oi ON oi.ProductID = p.ProductID
GROUP BY p.ProductID, p.SKU, p.Name, cat.Name, p.Price, p.CostPrice;
GO
```

### 2.3. Представление с SCHEMABINDING и индексированное представление

```sql
-- SCHEMABINDING запрещает менять базовые столбцы и позволяет создать
-- индексированное представление (материализованные данные).

-- Индексированное представление требует:
-- 1) WITH SCHEMABINDING
-- 2) Все ссылки — через схему.имя
-- 3) Нельзя использовать: *, OUTER JOIN (в некоторых случаях), агрегатные функции вне SUM/COUNT_BIG
-- 4) Первый индекс должен быть уникальным кластерным

CREATE OR ALTER VIEW sales.vCategoryRevenue
WITH SCHEMABINDING
AS
SELECT 
    cat.CategoryID,
    cat.Name AS CategoryName,
    COUNT_BIG(*) AS OrdersCount,
    SUM(oi.Quantity * oi.UnitPrice * (1 - oi.Discount/100.0)) AS Revenue
FROM ref.Categories cat
JOIN ref.Products p    ON p.CategoryID = cat.CategoryID
JOIN sales.OrderItems oi ON oi.ProductID = p.ProductID
GROUP BY cat.CategoryID, cat.Name;
GO

-- Создаём уникальный кластерный индекс — это «материализует» представление
CREATE UNIQUE CLUSTERED INDEX IX_vCategoryRevenue
ON sales.vCategoryRevenue(CategoryID);
GO

-- Теперь запросы к представлению читаются из индекса, а не пересчитываются
SELECT * FROM sales.vCategoryRevenue WITH (NOEXPAND);
```

### 2.4. Обновляемое представление и INSTEAD OF триггер

```sql
-- Представление содержит вычисляемые столбцы — напрямую обновлять нельзя.
CREATE OR ALTER VIEW sales.vOrderItemDetails
AS
SELECT 
    oi.OrderItemID,
    oi.OrderID,
    oi.ProductID,
    p.Name AS ProductName,
    oi.Quantity,
    oi.UnitPrice,
    oi.Discount,
    oi.Quantity * oi.UnitPrice * (1 - oi.Discount/100.0) AS LineTotal
FROM sales.OrderItems oi
JOIN ref.Products p ON p.ProductID = oi.ProductID;
GO

-- Попытка прямого UPDATE вызовет ошибку, т.к. ProductName и LineTotal — вычисляемые.
-- Решение — INSTEAD OF триггер на представлении:
CREATE OR ALTER TRIGGER sales.tr_vOrderItemDetails_Update
ON sales.vOrderItemDetails
INSTEAD OF UPDATE
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE oi
    SET 
        oi.Quantity   = i.Quantity,
        oi.UnitPrice  = i.UnitPrice,
        oi.Discount   = i.Discount
    FROM sales.OrderItems oi
    JOIN inserted i ON i.OrderItemID = oi.OrderItemID;
END;
GO

-- Обновление через представление
UPDATE sales.vOrderItemDetails
SET Discount = 15
WHERE OrderItemID = 3;
```

### 3.1. AFTER INSERT — автоматический пересчёт суммы заказа

```sql
-- При вставке новой позиции заказа пересчитываем TotalAmount.
CREATE OR ALTER TRIGGER sales.tr_OrderItems_AfterInsert
ON sales.OrderItems
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE o
    SET TotalAmount = x.NewTotal
    FROM sales.Orders o
    JOIN (
        SELECT oi.OrderID, 
               SUM(oi.Quantity * oi.UnitPrice * (1 - oi.Discount/100.0)) AS NewTotal
        FROM sales.OrderItems oi
        WHERE oi.OrderID IN (SELECT DISTINCT OrderID FROM inserted)
        GROUP BY oi.OrderID
    ) x ON x.OrderID = o.OrderID;
END;
GO

-- Проверка
INSERT INTO sales.OrderItems (OrderID, ProductID, Quantity, UnitPrice, Discount)
VALUES (10, 5, 1, 149990, 0);
SELECT OrderID, TotalAmount FROM sales.Orders WHERE OrderID = 10;
```

### 3.4. INSTEAD OF INSERT — валидация входящих данных

```sql
CREATE OR ALTER TRIGGER sales.tr_Orders_InsteadOfInsert
ON sales.Orders
INSTEAD OF INSERT
AS
BEGIN
    SET NOCOUNT ON;

    -- 1. Отклоняем заказы с отрицательной суммой
    IF EXISTS (SELECT 1 FROM inserted WHERE TotalAmount < 0)
    BEGIN
        THROW 50002, N'Сумма заказа не может быть отрицательной.', 1;
    END;

    -- 2. Отклоняем заказы с несуществующим клиентом
    IF EXISTS (
        SELECT 1 FROM inserted i
        WHERE NOT EXISTS (SELECT 1 FROM ref.Customers c WHERE c.CustomerID = i.CustomerID)
    )
    BEGIN
        THROW 50003, N'Клиент не найден.', 1;
    END;

    -- 3. Вставляем реальные данные со статусом по умолчанию
    INSERT INTO sales.Orders (CustomerID, EmployeeID, OrderDate, Status, TotalAmount)
    SELECT 
        CustomerID,
        EmployeeID,
        ISNULL(OrderDate, SYSDATETIME()),
        ISNULL(Status, N'новый'),
        ISNULL(TotalAmount, 0)
    FROM inserted;
END;
GO
```

## Часть 4. ТРАНЗАКЦИИ — расширенные сценарии

### 4.1. Явная транзакция с XACT_ABORT и THROW

```sql
-- XACT_ABORT ON автоматически откатывает транзакцию при ошибке.
-- Это проще, чем TRY/CATCH для «простых» случаев, но TRY/CATCH гибче.
SET XACT_ABORT ON;

BEGIN TRY
    BEGIN TRAN;

    -- Списание товара со склада
    UPDATE ref.Products 
    SET StockQty = StockQty - 2
    WHERE ProductID = 3 AND StockQty >= 2;

    IF @@ROWCOUNT = 0
        THROW 51000, N'Недостаточно товара на складе.', 1;

    -- Создание заказа
    INSERT INTO sales.Orders (CustomerID, EmployeeID, Status)
    VALUES (1, 5, N'новый');

    DECLARE @oid INT = SCOPE_IDENTITY();

    INSERT INTO sales.OrderItems (OrderID, ProductID, Quantity, UnitPrice)
    VALUES (@oid, 3, 2, 49990);

    COMMIT;
    PRINT N'Заказ оформлен.';
END TRY
BEGIN CATCH
    IF XACT_STATE() <> 0
        ROLLBACK;

    PRINT CONCAT(N'Ошибка: ', ERROR_MESSAGE(), 
                 N' (код ', ERROR_NUMBER(), 
                 N', строка ', ERROR_LINE(), N')');
END CATCH;

SET XACT_ABORT OFF;
```

### 4.2. Вложенные транзакции и SAVE TRANSACTION

```sql
-- SQL Server не поддерживает «настоящие» вложенные транзакции.
-- COMMIT внутренней транзакции не фиксирует данные, только уменьшает @@TRANCOUNT.
-- ROLLBACK внешней откатывает всё.

BEGIN TRAN Outer1;
    PRINT CONCAT('TRANCOUNT = ', @@TRANCOUNT);  -- 1

    INSERT INTO ref.Categories (Name) VALUES (N'Тест A');

    SAVE TRAN SavePointA;   -- точка сохранения

    BEGIN TRAN Inner2;
        PRINT CONCAT('TRANCOUNT = ', @@TRANCOUNT);  -- 2
        INSERT INTO ref.Categories (Name) VALUES (N'Тест B');

        -- Откат только к точке сохранения: удалит Тест B, но оставит Тест A
        ROLLBACK TRAN SavePointA;
        PRINT CONCAT('После отката к SavePointA, TRANCOUNT = ', @@TRANCOUNT); -- 2

    COMMIT TRAN Inner2;   -- уменьшает TRANCOUNT до 1, но данные не фиксируются
    PRINT CONCAT('После COMMIT Inner2, TRANCOUNT = ', @@TRANCOUNT); -- 1

COMMIT TRAN Outer1;       -- реальная фиксация
PRINT CONCAT('После COMMIT Outer1, TRANCOUNT = ', @@TRANCOUNT); -- 0
```

### 4.3. Обработка deadlock и повторная попытка

```sql
-- Deadlock (взаимная блокировка) — типичная ситуация в многопользовательской среде.
-- SQL Server убивает одну из транзакций с ошибкой 1205.
-- Правильный подход — повторять транзакцию.

CREATE OR ALTER PROCEDURE sales.uspPlaceOrder
    @CustomerID INT,
    @EmployeeID INT,
    @ProductID  INT,
    @Qty        INT
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    DECLARE @retry INT = 0, @maxRetry INT = 5;

    WHILE @retry <= @maxRetry
    BEGIN
        BEGIN TRY
            BEGIN TRAN;

            UPDATE ref.Products
            SET StockQty = StockQty - @Qty
            WHERE ProductID = @ProductID AND StockQty >= @Qty;

            IF @@ROWCOUNT = 0
                THROW 51001, N'Недостаточно товара.', 1;

            INSERT INTO sales.Orders (CustomerID, EmployeeID, Status)
            VALUES (@CustomerID, @EmployeeID, N'новый');

            COMMIT;
            RETURN;   -- успех
        END TRY
        BEGIN CATCH
            IF XACT_STATE() <> 0 ROLLBACK;

            -- 1205 = deadlock victim
            IF ERROR_NUMBER() = 1205 AND @retry < @maxRetry
            BEGIN
                SET @retry += 1;
                WAITFOR DELAY '00:00:00.200';  -- маленькая задержка перед повтором
                CONTINUE;
            END;

            THROW;
        END CATCH;
    END
END
GO
```

## Часть 5. КОМБИНИРОВАННЫЕ СЦЕНАРИИ

### 5.1. Хранимая процедура оформления заказа (транзакция + триггер + представление)

```sql
CREATE OR ALTER PROCEDURE sales.uspCreateOrder
    @CustomerID INT,
    @EmployeeID INT,
    @ItemsJson  NVARCHAR(MAX)   -- JSON вида [{"ProductID":1,"Qty":2,"Discount":5}]
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY
        BEGIN TRAN;

        -- 1. Создаём заказ
        INSERT INTO sales.Orders (CustomerID, EmployeeID, Status, TotalAmount)
        VALUES (@CustomerID, @EmployeeID, N'новый', 0);

        DECLARE @OrderID INT = SCOPE_IDENTITY();

        -- 2. Разбираем JSON и добавляем позиции
        INSERT INTO sales.OrderItems (OrderID, ProductID, Quantity, UnitPrice, Discount)
        SELECT 
            @OrderID,
            j.ProductID,
            j.Qty,
            p.Price,             -- цена берётся из справочника на момент продажи
            ISNULL(j.Discount, 0)
        FROM OPENJSON(@ItemsJson)
        WITH (
            ProductID INT   '$.ProductID',
            Qty       INT   '$.Qty',
            Discount  DECIMAL(5,2) '$.Discount'
        ) j
        JOIN ref.Products p ON p.ProductID = j.ProductID;

        -- 3. Проверяем наличие товара
        IF EXISTS (
            SELECT 1
            FROM sales.OrderItems oi
            JOIN ref.Products p ON p.ProductID = oi.ProductID
            WHERE oi.OrderID = @OrderID AND oi.Quantity > p.StockQty
        )
            THROW 52000, N'Недостаточно товара на складе.', 1;

        -- 4. Списываем остатки
        UPDATE p
        SET p.StockQty = p.StockQty - x.TotalQty
        FROM ref.Products p
        JOIN (
            SELECT ProductID, SUM(Quantity) AS TotalQty
            FROM sales.OrderItems
            WHERE OrderID = @OrderID
            GROUP BY ProductID
        ) x ON x.ProductID = p.ProductID;

        -- 5. Триггер sales.tr_OrderItems_AfterInsert уже пересчитал TotalAmount

        COMMIT;

        -- 6. Возвращаем сводку через представление
        SELECT * FROM sales.vOrderSummary WHERE OrderID = @OrderID;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0 ROLLBACK;
        THROW;
    END CATCH
END
GO

-- Пример вызова
EXEC sales.uspCreateOrder
    @CustomerID = 1,
    @EmployeeID = 5,
    @ItemsJson = N'[
        {"ProductID": 1, "Qty": 1, "Discount": 5},
        {"ProductID": 3, "Qty": 2, "Discount": 0}
    ]';
```

### 5.2. Отчёт: динамика продаж по категориям с оконными функциями и представлением

```sql
WITH Monthly AS (
    SELECT 
        cat.CategoryID,
        cat.Name AS CategoryName,
        DATEFROMPARTS(YEAR(o.OrderDate), MONTH(o.OrderDate), 1) AS MonthStart,
        SUM(oi.Quantity * oi.UnitPrice * (1 - oi.Discount/100.0)) AS Revenue
    FROM sales.Orders o
    JOIN sales.OrderItems oi ON oi.OrderID = o.OrderID
    JOIN ref.Products p      ON p.ProductID = oi.ProductID
    JOIN ref.Categories cat  ON cat.CategoryID = p.CategoryID
    WHERE o.Status <> N'отменён'
    GROUP BY cat.CategoryID, cat.Name,
             DATEFROMPARTS(YEAR(o.OrderDate), MONTH(o.OrderDate), 1)
)
SELECT 
    CategoryName,
    MonthStart,
    Revenue,
    -- Накопительный итог по категории
    SUM(Revenue) OVER (PARTITION BY CategoryID ORDER BY MonthStart
                       ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS RunningTotal,
    -- Доля месяца в общем итоге по категории
    Revenue / SUM(Revenue) OVER (PARTITION BY CategoryID) * 100 AS PctInCategory,
    -- Место категории по обороту в конкретном месяце
    RANK() OVER (PARTITION BY MonthStart ORDER BY Revenue DESC) AS RankInMonth
FROM Monthly
ORDER BY MonthStart, RankInMonth;
```

### 5.3. Soft Delete через INSTEAD OF DELETE + представление

```sql
-- Добавляем признак «удалён» к товарам
ALTER TABLE ref.Products ADD IsDeleted BIT NOT NULL DEFAULT 0;
GO

-- Представление «только живые товары»
CREATE OR ALTER VIEW ref.vActiveProducts
AS
SELECT * FROM ref.Products WHERE IsDeleted = 0;
GO

-- Триггер на представлении: DELETE → soft delete
CREATE OR ALTER TRIGGER ref.tr_vActiveProducts_Delete
ON ref.vActiveProducts
INSTEAD OF DELETE
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE p
    SET IsDeleted = 1, IsActive = 0
    FROM ref.Products p
    JOIN deleted d ON d.ProductID = p.ProductID;
END;
GO

-- Клиентский код работает так, будто удаляет:
DELETE FROM ref.vActiveProducts WHERE ProductID = 10;

-- Проверяем: товар остался, но помечен удалённым
SELECT ProductID, Name, IsDeleted FROM ref.Products WHERE ProductID = 10;
```

### 5.4. Очередь обработки заказов с блокировками READPAST

```sql
-- Несколько воркеров параллельно берут заказы из очереди «новый».
-- READPAST позволяет каждому взять свою строку, не дожидаясь других.
CREATE OR ALTER PROCEDURE sales.uspPickNextNewOrder
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRAN;

    DECLARE @OrderID INT;

    SELECT TOP (1) @OrderID = OrderID
    FROM sales.Orders WITH (UPDLOCK, ROWLOCK, READPAST)
    WHERE Status = N'новый'
    ORDER BY OrderDate;

    IF @OrderID IS NULL
    BEGIN
        COMMIT;
        RETURN;
    END;

    UPDATE sales.Orders
    SET Status = N'оплачен'
    WHERE OrderID = @OrderID;

    COMMIT;

    SELECT @OrderID AS PickedOrderID;
END
GO
```

## Итоговая шпаргалка

| Объект | Когда применять | Ключевые моменты |
|---|---|---|
| **Оконные функции** | Ранжирование, накопительные итоги, доли, сравнение с соседними строками | Нельзя в `WHERE`/`HAVING`; использовать `PARTITION BY`, явные рамки `ROWS`/`RANGE` |
| **Представления** | Инкапсуляция запросов, безопасность, упрощение отчётов | `SCHEMABINDING` + уникальный кластерный индекс = материализация; `INSTEAD OF` триггер делает обновляемым |
| **Триггеры** | Аудит, автоматизация расчётов, защита данных | Всегда работают с `inserted`/`deleted`; могут быть каскадными; `INSTEAD OF` для сложной логики |
| **Транзакции** | Атомарные операции, многопользовательская работа | `TRY/CATCH` + `XACT_ABORT ON` + `XACT_STATE()`; избегать длинных транзакций; повторять при deadlock 1205 |

Эта база данных и примеры полностью самодостаточны: создайте `ShopDB`, выполните скрипты по порядку — и вы получите работающую песочницу для отработки всех конструкций SQL Server.
